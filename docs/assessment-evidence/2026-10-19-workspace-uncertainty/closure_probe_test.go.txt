package workspace_test

import (
	"bytes"
	"context"
	"errors"
	"os"
	"path/filepath"
	"testing"

	"github.com/chester-hill-solutions/stow-s3/internal/storage"
	workspace "github.com/chester-hill-solutions/stow-s3/internal/storage/workspace"
)

// Temporary red probe. It asserts only what the pre-fix code can compile, so
// the failure it reports is the product's answer and not a build error.
func TestClosureWorkspaceUnreadable(t *testing.T) {
	if os.Geteuid() == 0 {
		t.Skip("root can read a file with no permissions, so this cannot be arranged here")
	}
	const bucket = "stow-closure-bucket"
	root := filepath.Join(t.TempDir(), "ws")
	store, err := workspace.New(workspace.Options{Root: root, Bucket: bucket, InitiallyOwned: true})
	if err != nil {
		t.Fatal(err)
	}
	defer store.Close()
	ctx := context.Background()
	if _, err := store.PutObject(ctx, bucket, "victim", bytes.NewReader([]byte("old")), storage.PutOptions{}); err != nil {
		t.Fatal(err)
	}
	victim := filepath.Join(root, "victim")
	if err := os.Chmod(victim, 0o000); err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = os.Chmod(victim, 0o600) })

	if _, err := store.HeadObject(ctx, bucket, "victim"); errors.Is(err, storage.ErrObjectNotFound) {
		t.Errorf("unreadable/head: %v, want the state reported as unknown, not as absence", err)
	}
	reader, _, err := store.GetObject(ctx, bucket, "victim")
	if errors.Is(err, storage.ErrObjectNotFound) {
		t.Errorf("unreadable/get: %v, want the state reported as unknown, not as absence", err)
	}
	if reader != nil {
		reader.Close()
	}
	if _, err := store.PutObject(ctx, bucket, "victim", bytes.NewReader([]byte("new")),
		storage.PutOptions{IfNoneMatch: "*"}); err == nil {
		t.Errorf("unreadable/create-only-write: the write was admitted over bytes the store could not read")
	}
	if err := os.Chmod(victim, 0o600); err != nil {
		t.Fatal(err)
	}
	after, err := os.ReadFile(victim)
	if err != nil {
		t.Fatal(err)
	}
	if string(after) != "old" {
		t.Errorf("unreadable/bytes: the object now holds %q, want the original bytes %q", after, "old")
	}
}
