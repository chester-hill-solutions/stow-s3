package stow_test

import (
	"context"
	"errors"
	"github.com/chester-hill-solutions/stow-s3/pkg/stow"
	"testing"
)

func TestReviewBufferedPutPreservesCommittedIdentity(t *testing.T) {
	ctx := context.Background()
	base, err := stow.Open(stow.Options{})
	if err != nil {
		t.Fatal(err)
	}
	defer base.Close()
	if err := base.CreateBucket(ctx, "probe"); err != nil {
		t.Fatal(err)
	}
	cause := errors.New("after local PUT")
	store := &copyPublicationStore{Store: base, writeErr: errors.Join(stow.ErrMutationCommitted, cause)}
	outer, err := stow.Open(stow.Options{Store: store, MaxBytes: 4, MaxObjects: 1})
	if err != nil {
		t.Fatal(err)
	}
	defer outer.Close()
	result, putErr := outer.PutObject(ctx, "probe", "target", []byte("body"), stow.PutOptions{})
	stored, readErr := base.GetObject(ctx, "probe", "target")
	if readErr != nil || string(stored.Data) != "body" {
		t.Fatalf("publication=%+v error=%v", stored, readErr)
	}
	if !errors.Is(putErr, stow.ErrMutationCommitted) || result.Key != "target" || result.ETag != stored.ETag {
		t.Fatalf("committed PUT result=%+v error=%v; actual=%+v usage=%+v", result, putErr, stored, outer.Usage())
	}
}
