import pathlib,re,json,statistics
p=pathlib.Path('/private/tmp/stow-save-body-perf-repeat/final-comparison');d={}
for f in sorted(p.glob('cpu*.log')):
 cpu,cycle,pos,version=re.match(r'cpu(\d+)-cycle(\d+)-position(\d+)-(\w+)',f.stem).groups()
 for m in re.finditer(r'(Benchmark\S+)\s+(\d+)\s+([\d.]+) ns/op\s+(\d+) B/op\s+(\d+) allocs/op',f.read_text()):
  name,n,ns,mem,alloc=m.groups();name=re.sub(r'-\d+$','',name)
  d.setdefault((int(cpu),name),{}).setdefault(version,[]).append({'cycle':int(cycle),'position':int(pos),'iterations':int(n),'ns':float(ns),'B':int(mem),'allocs':int(alloc),'source':f.name})
rows=[]
for (cpu,name),versions in sorted(d.items()):
 assert set(versions)=={'original','committed','phase'} and all(len(x)==6 for x in versions.values())
 summary={v:{'median':statistics.median(s['ns'] for s in samples),'min':min(s['ns'] for s in samples),'max':max(s['ns'] for s in samples),'bytes':sorted(set(s['B'] for s in samples)),'allocs':sorted(set(s['allocs'] for s in samples))} for v,samples in versions.items()}
 changes={v:100*(summary['phase']['median']/summary[v]['median']-1) for v in ['original','committed']}
 rows.append({'cpu':cpu,'case':name,'samples':versions,'summary':summary,'phase_change_pct_vs':changes})
 print(cpu,name,{v:s['median'] for v,s in summary.items()},{v:round(x,2) for v,x in changes.items()},'phase range',summary['phase']['min'],summary['phase']['max'])
assert len(rows)==22
(p/'measurements.json').write_text(json.dumps(rows,indent=2)+'\n')
