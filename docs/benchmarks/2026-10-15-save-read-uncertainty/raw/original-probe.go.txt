package runtime

import (
	"context"
	"errors"
	"io"
	"testing"

	"github.com/chester-hill-solutions/stow-s3/internal/storage"
)

type reviewSaveReadStore struct {
	*storage.MemoryStore
	readErr error
}

func (s *reviewSaveReadStore) GetObject(ctx context.Context, bucket, key string) (io.ReadCloser, *storage.ObjectMeta, error) {
	if s.readErr != nil {
		return nil, nil, s.readErr
	}
	return s.MemoryStore.GetObject(ctx, bucket, key)
}

func TestReviewReadForSaveAmbiguousAbsence(t *testing.T) {
	for _, order := range []string{"marker-first", "missing-first", "wrapped-marker-first", "wrapped-missing-first"} {
		t.Run(order, func(t *testing.T) {
			ctx := context.Background()
			store := &reviewSaveReadStore{MemoryStore: storage.NewMemoryStore()}
			instance, err := OpenWithStore(Options{Backend: BackendMemory, MaxBytes: 8, MaxObjects: 1}, store, nil)
			if err != nil {
				t.Fatal(err)
			}
			t.Cleanup(func() { _ = instance.Close() })
			if err := instance.CreateBucket(ctx, "probe"); err != nil {
				t.Fatal(err)
			}
			before, err := instance.PutObject(ctx, "probe", "target", []byte("old"), PutOptions{})
			if err != nil {
				t.Fatal(err)
			}
			store.readErr = objectHeadAmbiguity(order)
			_, condition, err := instance.ReadForSave(ctx, "probe", "target")
			if err != store.readErr {
				t.Errorf("unknown read reported success or lost error: %v want %v", err, store.readErr)
			}
			if condition.ObservedAbsence() {
				t.Errorf("unknown read fabricated authoritative absence for existing object")
			}
			store.readErr = nil
			if condition.ObservedAbsence() {
				result, err := instance.SaveObject(ctx, "probe", "target", []byte("new"), SaveOptions{Condition: condition})
				if !errors.Is(err, storage.ErrSaveConflict) || result.Outcome != SaveNotCommitted {
					t.Fatalf("backend absence guard failed: result=%+v error=%v", result, err)
				}
			}
			object, healthy, err := instance.ReadForSave(ctx, "probe", "target")
			if err != nil || healthy.ObservedAbsence() || string(object.Data) != "old" || object.VersionID != before.VersionID {
				t.Fatalf("healthy observation changed: object=%+v absent=%t error=%v", object, healthy.ObservedAbsence(), err)
			}
			result, err := instance.SaveObject(ctx, "probe", "target", []byte("new"), SaveOptions{Condition: healthy})
			if err != nil || result.Outcome != SaveCommitted {
				t.Fatalf("healthy retry: result=%+v error=%v", result, err)
			}
		})
	}
}
