type reviewOpeningErrorStore struct{
 *storage.MemoryStore
 path string
 failure error
 reader *os.File
 opens,closes int
}
func(s *reviewOpeningErrorStore)GetObject(ctx context.Context,bucket,key string)(io.ReadCloser,*storage.ObjectMeta,error){
 meta,err:=s.MemoryStore.HeadObject(ctx,bucket,key);if err!=nil{return nil,nil,err}
 file,err:=os.Open(s.path);if err!=nil{return nil,nil,err}
 s.reader=file;s.opens++
 return &reviewOpeningErrorReader{File:file,owner:s},meta,s.failure
}
type reviewOpeningErrorReader struct{*os.File;owner *reviewOpeningErrorStore}
func(r *reviewOpeningErrorReader)Close()error{r.owner.closes++;return r.File.Close()}
type reviewOpeningRangeErrorStore struct{*reviewOpeningErrorStore}
func(s *reviewOpeningRangeErrorStore)OpenObjectRange(ctx context.Context,bucket,key string,_ storage.ObjectRange)(io.ReadCloser,*storage.ObjectMeta,error){return s.GetObject(ctx,bucket,key)}

func TestReviewOpeningErrorDropsReader(t *testing.T){
 for _,mode:=range []string{"stream","get","guarded","adapter","fallback-range","native-range"}{
  t.Run(mode,func(t *testing.T){
   f,_:=saveBodyFixture(t)
   file:=filepath.Join(t.TempDir(),"body")
   if err:=os.WriteFile(file,[]byte("old"),0600);err!=nil{t.Fatal(err)}
   failure:=errors.New("provider opening failed")
   store:=&reviewOpeningErrorStore{MemoryStore:f.store.MemoryStore,path:file,failure:failure}
   f.instance.store=store
   if mode=="native-range"{f.instance.store=&reviewOpeningRangeErrorStore{store}}
   t.Cleanup(func(){if store.reader!=nil{_ = store.reader.Close()}})
   usage:=f.instance.Usage()
   var err error
   if mode=="native-range" || mode=="fallback-range"{
    reader,_,openErr:=f.instance.OpenObjectRange(context.Background(),"probe","target",storage.ObjectRange{Length:1});err=openErr
    if reader!=nil{_ = reader.Close()}
   }else{_,_,err=observeExcessRead(f.instance,mode)}
   if err!=failure{t.Fatalf("original opening error changed: %v",err)}
   if store.opens!=1{t.Fatal(fmt.Sprintf("opens=%d",store.opens))}
   if store.closes!=1{
    _,statErr:=store.reader.Stat()
    t.Errorf("opening error discarded reader: closes=%d file_still_open=%t",store.closes,statErr==nil)
   }
   if f.instance.Usage()!=usage{t.Fatal("opening refusal changed usage")}
   _ = store.reader.Close();store.reader=nil
   f.instance.store=store.MemoryStore
   verifySaveReadRecovery(t,f)
  })
 }
}
