func (i *Instance) reviewBaselineOpen(ctx context.Context, bucket, key string) (io.ReadCloser, Object, error) {
	return i.reviewBaselineOpenImpl(ctx, bucket, key, nil)
}

func (i *Instance) reviewBaselineRange(ctx context.Context, bucket, key string, span storage.ObjectRange) (io.ReadCloser, Object, error) {
	if span.Offset < 0 || span.Length < 0 {
		return nil, Object{}, fmt.Errorf("object range offset and length must not be negative")
	}
	return i.reviewBaselineOpenImpl(ctx, bucket, key, &span)
}

func (i *Instance) reviewBaselineOpenImpl(ctx context.Context, bucket, key string, span *storage.ObjectRange) (io.ReadCloser, Object, error) {
	if err := i.checkContext(ctx); err != nil {
		return nil, Object{}, err
	}
	if err := i.checkResource(authority.ObjectRead, policy.Object(bucket, key)); err != nil {
		return nil, Object{}, err
	}
	i.mu.RLock()
	defer i.mu.RUnlock()
	if err := i.checkOpen(); err != nil {
		return nil, Object{}, err
	}
	reader, meta, err := i.reviewBaselineStoreRange(ctx, bucket, key, span)
	if err != nil {
		return nil, Object{}, err
	}
	stream := &reviewBaselineStream{instance: i, context: ctx, bucket: bucket, key: key, reader: reader}
	if seeker, ok := reader.(io.Seeker); ok {
		return &reviewBaselineSeekableStream{reviewBaselineStream: stream, seeker: seeker}, objectFromMeta(meta, nil), nil
	}
	return stream, objectFromMeta(meta, nil), nil
}

func (i *Instance) reviewBaselineStoreRange(ctx context.Context, bucket, key string, span *storage.ObjectRange) (io.ReadCloser, *storage.ObjectMeta, error) {
	if ranged, ok := i.store.(storage.RangeStore); ok && span != nil {
		return ranged.OpenObjectRange(ctx, bucket, key, *span)
	}
	reader, meta, err := i.store.GetObject(ctx, bucket, key)
	if err != nil || span == nil {
		return reader, meta, err
	}
	if span.Offset > meta.Size || span.Length > meta.Size-span.Offset {
		_ = reader.Close()
		return nil, nil, fmt.Errorf("object range exceeds object size")
	}
	var skipErr error
	if seeker, ok := reader.(io.Seeker); ok {
		_, skipErr = seeker.Seek(span.Offset, io.SeekStart)
	} else {
		_, skipErr = io.CopyN(io.Discard, reader, span.Offset)
	}
	if skipErr != nil {
		_ = reader.Close()
		return nil, nil, skipErr
	}
	return &reviewBaselineLimitedReader{Reader: io.LimitReader(reader, span.Length), closer: reader}, meta, nil
}

type reviewBaselineLimitedReader struct {
	io.Reader
	closer io.Closer
}

func (r *reviewBaselineLimitedReader) Close() error { return r.closer.Close() }

type reviewBaselineStream struct {
	mu       sync.Mutex
	instance *Instance
	context  context.Context
	bucket   string
	key      string
	reader   io.ReadCloser
	closed   bool
}

func (r *reviewBaselineStream) Read(buffer []byte) (int, error) {
	r.mu.Lock()
	defer r.mu.Unlock()
	if r.closed {
		return 0, io.ErrClosedPipe
	}
	i := r.instance
	i.mu.RLock()
	defer i.mu.RUnlock()
	if err := r.authorizeLocked(); err != nil {
		_ = r.closeLocked()
		return 0, err
	}
	n, err := r.reader.Read(buffer)
	if err != nil {
		_ = r.closeLocked()
	}
	return n, err
}

func (r *reviewBaselineStream) closeLocked() error {
	if r.closed {
		return nil
	}
	r.closed = true
	return r.reader.Close()
}

func (r *reviewBaselineStream) Close() error {
	r.mu.Lock()
	defer r.mu.Unlock()
	return r.closeLocked()
}

func (r *reviewBaselineStream) authorizeLocked() error {
	i := r.instance
	if err := i.checkContext(r.context); err != nil {
		return err
	}
	if err := i.checkOpen(); err != nil {
		return err
	}
	return i.checkResource(authority.ObjectRead, policy.Object(r.bucket, r.key))
}

type reviewBaselineSeekableStream struct {
	*reviewBaselineStream
	seeker io.Seeker
}

func (r *reviewBaselineSeekableStream) Seek(offset int64, whence int) (int64, error) {
	r.mu.Lock()
	defer r.mu.Unlock()
	if r.closed {
		return 0, io.ErrClosedPipe
	}
	r.instance.mu.RLock()
	defer r.instance.mu.RUnlock()
	if err := r.authorizeLocked(); err != nil {
		_ = r.closeLocked()
		return 0, err
	}
	return r.seeker.Seek(offset, whence)
}

func BenchmarkStreamMatched(b *testing.B) {
	for _, item := range streamBenchmarkCases() {
		if item.adapter || item.short {
			continue
		}
		b.Run(item.name, func(b *testing.B) {
			for _, label := range []string{"before-a", "after-a", "after-b", "before-b"} {
				b.Run(label, func(b *testing.B) {
					opener := openStreamBenchmark
					if strings.HasPrefix(label, "before") {
						opener = reviewBaselineStreamOpener
					}
					benchmarkStreamCompleteness(b, item, opener)
				})
			}
		})
	}
}
func reviewBaselineStreamOpener(instance *Instance, item streamBenchmarkCase) (io.ReadCloser, error) {
	if item.ranged {
		reader, _, err := instance.reviewBaselineRange(context.Background(), "probe", "target", storage.ObjectRange{Offset: 1, Length: int64(item.size - 1)})
		return reader, err
	}
	reader, _, err := instance.reviewBaselineOpen(context.Background(), "probe", "target")
	return reader, err
}

func (i *Instance) reviewBaselineGet(ctx context.Context, bucket, key string) (Object, error) {
	reader, object, err := i.reviewBaselineOpen(ctx, bucket, key)
	if err != nil {
		return Object{}, err
	}
	if err := readOpenedObject(reader, &object); err != nil {
		return Object{}, err
	}
	return object, nil
}

func (i *Instance) reviewBaselineReadForSave(ctx context.Context, bucket, key string) (Object, SaveCondition, error) {
	if err := i.checkContext(ctx); err != nil {
		return Object{}, SaveCondition{}, err
	}
	if err := i.checkResource(authority.ObjectRead, policy.Object(bucket, key)); err != nil {
		return Object{}, SaveCondition{}, err
	}
	if !i.SupportsGuardedSaves() {
		return Object{}, SaveCondition{}, ErrGuardedSavesUnsupported
	}
	reader, object, err := i.reviewBaselineOpen(ctx, bucket, key)
	condition := SaveCondition{owner: i, bucket: bucket, key: key}
	if errors.Is(err, storage.ErrObjectNotFound) && !errors.Is(err, storage.ErrMutationCommitted) {
		condition.guard.Absent = true
		return Object{Bucket: bucket, Key: key}, condition, nil
	}
	if err != nil {
		return Object{}, SaveCondition{}, err
	}
	err = readOpenedObject(reader, &object)
	if err != nil {
		return Object{}, SaveCondition{}, err
	}
	condition.guard.Fingerprint = storage.ObjectFingerprint(storage.ObjectMeta{
		VersionID: object.VersionID, ContentType: object.ContentType,
		Metadata: object.Metadata, ChecksumAlgorithm: object.ChecksumAlgorithm, ChecksumValue: object.ChecksumValue,
	}, object.Data)
	return object, condition, nil
}

type reviewStreamPairedCase struct {
	streamBenchmarkCase
	mode  string
	count int
}

func TestReviewStreamPairedCosts(t *testing.T) {
	cases := []reviewStreamPairedCase{}
	for _, item := range streamBenchmarkCases() {
		if item.adapter || item.short {
			continue
		}
		n := 10000
		if item.size == 3 {
			n = 50000
		}
		cases = append(cases, reviewStreamPairedCase{item, "stream", n})
	}
	cases = append(cases, reviewStreamPairedCase{streamBenchmarkCase{name: "get-tiny", size: 3}, "get", 50000}, reviewStreamPairedCase{streamBenchmarkCase{name: "get-64KiB", size: 64 << 10}, "get", 5000}, reviewStreamPairedCase{streamBenchmarkCase{name: "save-64KiB", size: 64 << 10}, "save", 1000})
	for _, item := range cases {
		reviewStreamPairedCaseRun(t, item)
	}
}
func reviewStreamPairedCaseRun(t *testing.T, item reviewStreamPairedCase) {
	instance, err := Open(Options{})
	if err != nil {
		t.Fatal(err)
	}
	defer instance.Close()
	ctx := context.Background()
	if err := instance.CreateBucket(ctx, "probe"); err != nil {
		t.Fatal(err)
	}
	if _, err := instance.PutObject(ctx, "probe", "target", bytes.Repeat([]byte("x"), item.size), PutOptions{ContentType: "text/plain", Metadata: map[string]string{"state": "original"}}); err != nil {
		t.Fatal(err)
	}
	readerBefore, _, err := instance.reviewBaselineOpen(ctx, "probe", "target")
	if err != nil {
		t.Fatal(err)
	}
	readerAfter, _, err := instance.OpenObject(ctx, "probe", "target")
	if err != nil {
		t.Fatal(err)
	}
	types := [2]reflect.Type{reflect.TypeOf(readerBefore), reflect.TypeOf(readerAfter)}
	if types[0].NumMethod() != types[1].NumMethod() {
		t.Fatal("different reader method sets")
	}
	for n := 0; n < types[0].NumMethod(); n++ {
		if types[0].Method(n).Name != types[1].Method(n).Name {
			t.Fatal("different reader methods")
		}
	}
	if _, ok := readerBefore.(io.WriterTo); !ok {
		t.Fatal("baseline erased WriterTo")
	}
	_ = readerBefore.Close()
	_ = readerAfter.Close()
	before := reviewStreamWork(instance, item, true)
	after := reviewStreamWork(instance, item, false)
	for range 100 {
		before()
		after()
	}
	for cycle := 0; cycle < 10; cycle++ {
		labels := []string{"before-a", "after-a", "after-b", "before-b"}
		if cycle%2 == 1 {
			labels = []string{"after-a", "before-a", "before-b", "after-b"}
		}
		for position, label := range labels {
			work := after
			if strings.HasPrefix(label, "before") {
				work = before
			}
			samples := []metrics.Sample{{Name: "/cpu/classes/gc/total:cpu-seconds"}}
			var m0, m1 stdruntime.MemStats
			stdruntime.ReadMemStats(&m0)
			metrics.Read(samples)
			gc0 := samples[0].Value.Float64()
			start := time.Now()
			for n := 0; n < item.count; n++ {
				work()
			}
			elapsed := time.Since(start).Nanoseconds()
			stdruntime.ReadMemStats(&m1)
			metrics.Read(samples)
			gc1 := samples[0].Value.Float64()
			row := map[string]interface{}{"case": item.name, "cpu": stdruntime.GOMAXPROCS(0), "cycle": cycle, "position": position, "label": label, "n": item.count, "ns": elapsed, "allocated_bytes": m1.TotalAlloc - m0.TotalAlloc, "allocs": m1.Mallocs - m0.Mallocs, "gc_cycles": m1.NumGC - m0.NumGC, "gc_pause_ns": m1.PauseTotalNs - m0.PauseTotalNs, "gc_cpu_seconds": gc1 - gc0}
			encoded, err := json.Marshal(row)
			if err != nil {
				t.Fatal(err)
			}
			t.Logf("PAIRED %s", encoded)
		}
	}
}
func reviewStreamWork(instance *Instance, item reviewStreamPairedCase, before bool) func() {
	ctx := context.Background()
	switch item.mode {
	case "get":
		if before {
			return func() {
				object, err := instance.reviewBaselineGet(ctx, "probe", "target")
				if err != nil || len(object.Data) != item.size {
					panic("baseline GET")
				}
			}
		}
		return func() {
			object, err := instance.GetObject(ctx, "probe", "target")
			if err != nil || len(object.Data) != item.size {
				panic("candidate GET")
			}
		}
	case "save":
		if before {
			return func() {
				object, condition, err := instance.reviewBaselineReadForSave(ctx, "probe", "target")
				if err != nil || len(object.Data) != item.size || condition.ObservedAbsence() {
					panic("baseline guarded read")
				}
			}
		}
		return func() {
			object, condition, err := instance.ReadForSave(ctx, "probe", "target")
			if err != nil || len(object.Data) != item.size || condition.ObservedAbsence() {
				panic("candidate guarded read")
			}
		}
	}
	opener := openStreamBenchmark
	if before {
		opener = reviewBaselineStreamOpener
	}
	buffer := make([]byte, item.chunk)
	return func() {
		reader, err := opener(instance, item.streamBenchmarkCase)
		if err != nil {
			panic(err)
		}
		if item.seek {
			if _, err := reader.(io.Seeker).Seek(int64(item.size/2), io.SeekStart); err != nil {
				panic(err)
			}
		}
		n, err := consumeCompletionStream(reader, buffer)
		if err != io.EOF || n != streamBenchmarkLength(item.streamBenchmarkCase) {
			panic("stream result changed")
		}
		if err := reader.Close(); err != nil {
			panic(err)
		}
	}
}

func (r *reviewBaselineStream) WriteTo(writer io.Writer) (int64, error) {
	buffer, _ := streamCopyBuffers.Get().(*[]byte)
	if buffer == nil {
		data := make([]byte, 128<<10)
		buffer = &data
	}
	defer streamCopyBuffers.Put(buffer)
	return io.CopyBuffer(struct{ io.Writer }{writer}, struct{ io.Reader }{r}, *buffer)
}
