import{readFile,writeFile,mkdir}from'node:fs/promises';import{spawnSync}from'node:child_process';import{createHash}from'node:crypto';import{join}from'node:path';import{loadavg}from'node:os';import{systemMetadata,cgroupResources}from'./resources.mjs';
if(process.platform!=='linux'||process.env.RAILWAY_SERVICE_ID!=='83497842-c51d-420e-8ba1-181589730ad4'||process.env.RAILWAY_ENVIRONMENT_ID!=='3cd200cb-c94f-4e4d-9335-0f6c1b7e9411'||!process.env.RAILWAY_DEPLOYMENT_ID)throw Error('Railway runner only');
const root=process.argv[2];if(!root?.startsWith('/results/'))throw Error('Results volume required');await mkdir(root,{recursive:false});
const save=(name,value)=>writeFile(join(root,name),typeof value==='string'?value:JSON.stringify(value,null,2)+'\n');
try{
 const start=await systemMetadata(),build=JSON.parse(await readFile('benchmark/build-identity.json'));
 if(start.cgroup.cpuMax!=='400000 100000'||start.cgroup.memoryMax!=='4294967296')throw Error('Unexpected resource limits');
 for(const [name,value]of Object.entries(build.binaries))if(createHash('sha256').update(await readFile('benchmark/'+name+'.test')).digest('hex')!==value.sha256)throw Error('Binary mismatch');
 const plan=[];for(const cpu of [1,4])for(let cycle=0;cycle<2;cycle++)for(const variant of cycle===0?['baseline','record','fixed']:['fixed','record','baseline'])plan.push({cpu,cycle,variant});
 await save('plan.json',{blocks:plan,benchtime:'200ms',count:2});const rows=[],resources=[];
 for(const[index,block]of plan.entries()){
  const before=await cgroupResources(),loadBefore=loadavg();
  const command=['benchmark/'+block.variant+'.test','-test.run=^$','-test.bench=^BenchmarkWorkspaceMultipartComparison$','-test.benchmem','-test.benchtime=200ms','-test.count=2','-test.cpu='+block.cpu];
  const q=spawnSync(command[0],command.slice(1),{encoding:'utf8',timeout:240000,maxBuffer:4194304});
  await save('block-'+index+'.log','COMMAND '+JSON.stringify(command)+'\n'+q.stdout+q.stderr);
  resources.push({block:index,before,after:await cgroupResources(),loadBefore,loadAfter:loadavg()});
  if(q.status!==0)throw Error('Block '+index+' failed: '+q.status+' '+q.error);
  for(const line of q.stdout.split('\n'))if(line.startsWith('Benchmark')){const tokens=line.trim().split(/\s+/),metrics={};for(let i=2;i<tokens.length-1;i+=2)metrics[tokens[i+1]]=Number(tokens[i]);rows.push({...block,block:index,name:tokens[0].replace(/-\d+$/,''),iterations:Number(tokens[1]),metrics})}
  await save('progress.json',{blocks:index+1,total:plan.length});console.log(JSON.stringify({block:index,...block,complete:true}));
 }
 if(rows.length!==144)throw Error('Expected 144 samples, got '+rows.length);
 await save('samples.json',rows);await save('block-resources.json',resources);await save('environment.json',{provider:'Railway',...build,start,end:await systemMetadata()});await save('run.json',{complete:true,samples:rows.length,railwayDeploymentId:process.env.RAILWAY_DEPLOYMENT_ID});
}catch(error){await save('run.json',{complete:false,error:error.stack});throw error}
