package workspace_test

import (
 "context"
 "errors"
 "os"
 "path/filepath"
 "strings"
 "testing"

 stowruntime "github.com/chester-hill-solutions/stow-s3/internal/runtime"
 "github.com/chester-hill-solutions/stow-s3/internal/storage"
 "github.com/chester-hill-solutions/stow-s3/internal/storage/workspace"
)

func TestNextWorkspaceOwnerRecovery(t *testing.T) {
 for _, damaged := range []bool{false,true} {
  name := "healthy"; if damaged { name = "damaged" }
  t.Run(name,func(t *testing.T) {
   ctx:=context.Background(); store,root:=newStore(t)
   upload,err:=store.CreateMultipartUpload(ctx,bucket,"target",storage.MultipartOptions{});if err!=nil {t.Fatal(err)}
   if _,err:=store.UploadPart(ctx,upload.UploadID,1,strings.NewReader("replacement"));err!=nil {t.Fatal(err)}
   if err:=store.Close();err!=nil {t.Fatal(err)}
   owner:=filepath.Join(workspace.InternalPath(root,"multipart",upload.UploadID),"upload.json")
   saved,err:=os.ReadFile(owner);if err!=nil {t.Fatal(err)}
   if damaged {if err:=os.WriteFile(owner,[]byte("{"),0600);err!=nil {t.Fatal(err)}}
   reopened,err:=workspace.New(workspace.Options{Root:root,Bucket:bucket});if err!=nil {t.Fatal(err)}
   instance,err:=stowruntime.OpenWithStore(stowruntime.Options{Backend:stowruntime.BackendWorkspace,MaxBytes:12,MaxObjects:1},reopened,nil)
   if err!=nil {reopened.Close();return}
   _,putErr:=instance.PutObject(ctx,bucket,"extra",[]byte("123456789012"),stowruntime.PutOptions{})
   if err:=instance.Close();err!=nil {t.Fatal(err)}
   if err:=os.WriteFile(owner,saved,0600);err!=nil {t.Fatal(err)}
   repaired,err:=workspace.New(workspace.Options{Root:root,Bucket:bucket});if err!=nil {t.Fatal(err)}
   defer repaired.Close()
   parts,err:=repaired.ListParts(ctx,upload.UploadID);if err!=nil||len(parts)!=1||parts[0].Size!=11 {t.Fatalf("retained part=%+v error=%v",parts,err)}
   t.Logf("damaged=%v extra12bytes error=%v retainedPartBytes=%d cap=12",damaged,putErr,parts[0].Size)
   if !errors.Is(putErr,stowruntime.ErrQuotaExceeded) {t.Fatalf("retained 11-byte part omitted: new 12-byte object accepted under cap12: %v",putErr)}
  })
 }
}
