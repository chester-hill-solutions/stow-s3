package runthrough

import (
	"bytes"
	"os"
	"path/filepath"
	"testing"
)

func TestFileOutboxUnchangedProvenanceDoesNotRewrite(t *testing.T) {
	path := filepath.Join(t.TempDir(), "outbox.json")
	outbox, err := NewFileOutbox(path)
	if err != nil {
		t.Fatal(err)
	}
	for _, state := range []UpstreamState{{ETag: "first"}, {Absent: true}, {Unknown: true}, {}} {
		if err := outbox.SaveUpstreamState("bucket", "key", state); err != nil {
			t.Fatal(err)
		}
		before, err := os.Stat(path)
		if err != nil {
			t.Fatal(err)
		}
		data, err := os.ReadFile(path)
		if err != nil {
			t.Fatal(err)
		}
		if err := outbox.SaveUpstreamState("bucket", "key", state); err != nil {
			t.Fatal(err)
		}
		after, err := os.Stat(path)
		if err != nil {
			t.Fatal(err)
		}
		saved, err := os.ReadFile(path)
		if err != nil {
			t.Fatal(err)
		}
		if !os.SameFile(before, after) || !bytes.Equal(data, saved) {
			t.Errorf("unchanged state rewrote journal: %+v", state)
		}
	}
}
