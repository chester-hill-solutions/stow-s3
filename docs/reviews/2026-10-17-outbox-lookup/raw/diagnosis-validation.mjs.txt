import{readFile,writeFile,mkdir,open,copyFile,chown}from'node:fs/promises';import{spawn}from'node:child_process';import{join,basename}from'node:path';import{createHash}from'node:crypto';import{systemMetadata}from'./resources.mjs';
if(process.platform!=='linux'||process.env.RAILWAY_SERVICE_ID!=='83497842-c51d-420e-8ba1-181589730ad4'||process.env.RAILWAY_ENVIRONMENT_ID!=='3cd200cb-c94f-4e4d-9335-0f6c1b7e9411'||!process.env.RAILWAY_DEPLOYMENT_ID)throw Error('Railway runner only');
const root=process.argv[2];if(!root?.startsWith('/results/'))throw Error('Results volume required');await mkdir(root,{recursive:false});await chown(root,10001,10001);const save=(name,value)=>writeFile(join(root,name),typeof value==='string'?value:JSON.stringify(value,null,2)+'\n');const config=JSON.parse(await readFile('benchmark/active-plan.json'));const results=[];
async function run(step){const log=await open(join(root,step.name+'.log'),'w');const start=Date.now();let result;try{const child=spawn('python3',['benchmark/reap_validation.py','runuser','-u','stow-validation','--',...step.args],{env:{...process.env,STOW_RESULT_ROOT:root,...step.env},stdio:['ignore',log.fd,log.fd]});result=await new Promise((resolve,reject)=>{child.once('error',reject);child.once('close',(code,signal)=>resolve({exitCode:code,signal}))});}finally{await log.close()}return{...step,...result,elapsedMs:Date.now()-start,passed:result.exitCode===(step.expected??0)};}
try{await save('plan.json',config);const before=await systemMetadata();
for(const step of config.steps){console.log('Starting '+step.name);results.push(await run(step));await save('checks.json',results);await save('progress.json',{completed:results.length,total:config.steps.length,last:results.at(-1)});console.log(JSON.stringify(results.at(-1)));}
const sources={};for(const name of config.sources){const data=await readFile(name);sources[name]=createHash('sha256').update(data).digest('hex');await copyFile(name,join(root,'source-'+basename(name)+'.txt'));}await save('environment.json',{provider:'Railway',start:before,end:await systemMetadata(),sourceSHA256:sources,sourceManifest:JSON.parse(await readFile('benchmark/validation-input-manifest.json'))});await save('run.json',{complete:true,success:results.every(r=>r.passed),railwayDeploymentId:process.env.RAILWAY_DEPLOYMENT_ID,checks:results.length,phase:config.phase});
}catch(error){await save('run.json',{complete:false,error:error.stack});throw error;}
