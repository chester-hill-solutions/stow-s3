import{readFile,writeFile,mkdir,open,copyFile,mkdtemp,statfs,chown,rm}from'node:fs/promises';
import{spawn}from'node:child_process';import{createHash}from'node:crypto';import{join}from'node:path';import{loadavg}from'node:os';
import{systemMetadata,cgroupResources}from'./resources.mjs';
export async function run(root,plan){
 if(process.platform!=='linux'||process.env.RAILWAY_SERVICE_ID!=='83497842-c51d-420e-8ba1-181589730ad4'||process.env.RAILWAY_ENVIRONMENT_ID!=='3cd200cb-c94f-4e4d-9335-0f6c1b7e9411'||!process.env.RAILWAY_DEPLOYMENT_ID)throw Error('Railway runner only');
 if(!root?.startsWith('/results/'))throw Error('Results volume required');await mkdir(root,{recursive:false});await chown(root,10001,10001);
 const save=(name,value)=>writeFile(join(root,name),typeof value==='string'?value:JSON.stringify(value,null,2)+'\n');const hash=x=>createHash('sha256').update(x).digest('hex');
 const tmp=await mkdtemp('/results/stow-perf-tmp-');await chown(tmp,10001,10001);
 async function command(name,args,extra={}){const log=await open(join(root,name+'.log'),'w');let result;try{const child=spawn('python3',['benchmark/reap_validation.py','runuser','-u','stow-validation','--',...args],{env:{...process.env,TMPDIR:tmp,...extra},stdio:['ignore',log.fd,log.fd]});result=await new Promise((resolve,reject)=>{child.once('error',reject);child.once('close',(exitCode,signal)=>resolve({exitCode,signal}))});}finally{await log.close()}if(result.exitCode!==0)throw Error(name+' failed '+JSON.stringify(result));return readFile(join(root,name+'.log'),'utf8')}
 try{
 const start=await systemMetadata();if(start.cgroup.cpuMax!=='400000 100000'||start.cgroup.memoryMax!=='4294967296')throw Error('Unexpected resource limits');
 const filesystem=await statfs(tmp);if(filesystem.type!==0xef53)throw Error('Expected ext4 results volume');
 await save('plan.json',plan);const variants=JSON.parse(await readFile('benchmark/perf-variants.json')),build={sources:{},binaries:{},variants},all=plan.families.flatMap(f=>plan.variants[f]);
 const formatFiles=all.flatMap(v=>variants[v].files.map(n=>'benchmark/_outbox-variants/'+v+'/'+n));if(formatFiles.length)await command('format-prototypes',['gofmt','-w',...formatFiles]);
 await mkdir(join(root,'bin'));await chown(join(root,'bin'),10001,10001);
 for(const variant of all){const cfg=variants[variant];if(cfg.alias){await copyFile(join(root,'bin',cfg.alias+'.test'),join(root,'bin',variant+'.test'));const bytes=await readFile(join(root,'bin',variant+'.test'));build.binaries[variant]={bytes:bytes.length,sha256:hash(bytes)};continue}const args=['go','test','-c','-buildvcs=false','-o',join(root,'bin',variant+'.test')],replace={};for(const name of cfg.files){const file='benchmark/_outbox-variants/'+variant+'/'+name;replace['/app/'+name]='/app/'+file;build.sources[file]=hash(await readFile(file));await copyFile(file,join(root,'source-'+variant+'-'+name.split('/').at(-1)+'.txt'))}if(cfg.files.length){const file=join(root,variant+'-overlay.json');await writeFile(file,JSON.stringify({Replace:replace}));args.push('-overlay='+file)}args.push(cfg.package);await command('build-'+variant,args);const bytes=await readFile(join(root,'bin',variant+'.test'));build.binaries[variant]={bytes:bytes.length,sha256:hash(bytes)}}
 for(const name of ['internal/runthrough/file_outbox.go','internal/runthrough/provenance.go','internal/runthrough/file_outbox_lookup_test.go','internal/runthrough/outbox.go','internal/runthrough/file_outbox_provenance_bench_test.go','internal/runthrough/file_outbox_provenance_test.go','internal/runthrough/file_outbox_fence_api_test.go','internal/runthrough/file_outbox_proto.go','internal/runthrough/file_outbox_proto_test.go','internal/runthrough/file_outbox_proto_bench_test.go','internal/runthrough/file_outbox_proto_selector_test.go','internal/runthrough/outboxpb/outbox.pb.go','internal/runthrough/outboxpb/outbox.proto'])build.sources[name]=hash(await readFile(name));
 await command('go-version',['go','version']);await command('go-env',['go','env','-json','GOOS','GOARCH','GOTOOLCHAIN','GOAMD64','CGO_ENABLED','GOFLAGS','GOEXPERIMENT']);await save('build.json',build);
 const blocks=[];for(const family of plan.families)for(const cpu of plan.cpu){const variants=plan.variants[family];if(plan.stepwise){for(const[cycle,order]of [variants,[...variants].reverse()].entries())for(const variant of order)blocks.push({family,cpu,cycle,variant})}else for(let cycle=0;cycle<2;cycle++){const[a,b]=variants;for(const variant of cycle===0?[a,b,b,a]:[b,a,a,b])blocks.push({family,cpu,cycle,variant})}}
 await save('blocks.json',blocks);const rows=[],resources=[];
 for(const[index,block]of blocks.entries()){
 const before=await cgroupResources(),loadBefore=loadavg();const pattern=plan.pattern??(block.family==='parts'?'^BenchmarkWorkspaceMultipartComparison$':'^BenchmarkRefreshVersionFallback$');
 const args=[join(root,'bin',block.variant+'.test'),'-test.run=^$','-test.bench='+pattern,'-test.benchmem','-test.benchtime='+plan.benchtime,'-test.count='+plan.count,'-test.cpu='+block.cpu];const output=await command('block-'+index,args);
 resources.push({block:index,before,after:await cgroupResources(),loadBefore,loadAfter:loadavg()});for(const line of output.split('\n'))if(line.startsWith('Benchmark')){const tokens=line.trim().split(/\s+/),metrics={};for(let i=2;i<tokens.length-1;i+=2)metrics[tokens[i+1]]=Number(tokens[i]);rows.push({...block,block:index,name:tokens[0].replace(/-\d+$/,''),iterations:Number(tokens[1]),metrics})}
 await save('samples.json',rows);await save('block-resources.json',resources);await save('progress.json',{completed:index+1,total:blocks.length,last:block});console.log(JSON.stringify({block:index,...block}));
 }
 if(rows.length!==blocks.reduce((n,b)=>n+(plan.caseCount??(b.family==='parts'?6:4))*plan.count,0))throw Error('Unexpected sample count '+rows.length);
 if(plan.profile){const binary=join(root,'bin',(plan.profileVariant??'parts-current')+'.test');await command('alloc-profile',[binary,'-test.run=^$','-test.bench=^BenchmarkWorkspaceMultipartComparison/parts=16/bytes=1048576/list$','-test.benchtime=1000x','-test.cpu=1','-test.memprofilerate=1','-test.memprofile='+join(root,'parts-alloc.pprof')]);await command('alloc-top',['go','tool','pprof','-top','-alloc_objects',binary,join(root,'parts-alloc.pprof')]);await command('escape',['go','test','./internal/storage/workspace','-run=^$','-gcflags=-m=2']);}
 await save('environment.json',{provider:'Railway',start,end:await systemMetadata(),filesystem:{type:filesystem.type,blockSize:filesystem.bsize},build,temporaryDirectory:tmp});await save('run.json',{complete:true,success:true,samples:rows.length,railwayDeploymentId:process.env.RAILWAY_DEPLOYMENT_ID,cohort:plan.cohort});
 await rm(tmp,{recursive:true});await save('cleanup.json',{removedTemporaryDirectory:tmp});
 }catch(error){await save('run.json',{complete:false,error:error.stack});throw error}
}
