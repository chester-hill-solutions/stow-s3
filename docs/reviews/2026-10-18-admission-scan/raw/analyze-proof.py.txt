from pathlib import Path
import json,statistics,re,hashlib
p=Path('/private/tmp/stow-admission-scan-2026-10-18');phase='proof';j=json.loads((p/(phase+'-start.json')).read_text());e=p/('export-'+j['id']);v=json.loads((e/'run.json').read_text());checks=json.loads((e/'checks.json').read_text());assert v['complete'] and v['success'] and all(x['passed']for x in checks) and len(checks)==11
for n in ['red-normal','red-race','prototype-normal','prototype-race']:
 t=(e/(n+'.log')).read_text();assert 'build failed'not in t and 'DATA RACE'not in t and '--- SKIP:'not in t
 for op in ['put','save','delete']:
  m=re.search(op+r' with 8 MiB of existing payload: (\d+) allocated bytes/op',t);assert m,(n,op)
  allocated=int(m.group(1));assert allocated>=8<<20,(n,op,allocated)
  if n.startswith('red'):
   assert allocated>12<<20 and '--- FAIL: TestCapacityMutationAllocation/'+op in t,(n,op,allocated)
   assert op+' repeated existing payload reads:'in t,(n,op)
  else:
   assert allocated<12<<20 and '--- PASS: TestCapacityMutationAllocation/'+op in t,(n,op,allocated)
rows=json.loads((e/'measurements.json').read_text());assert len(rows['observations'])==144 and all(r['iterations']==10 for r in rows['observations'])
analysis=[]
for name in sorted({r['name']for r in rows['observations']}):
 cell={'name':name,'labels':{}}
 for label in ['A','B','C','D']:
  values=[r for r in rows['observations']if r['name']==name and r['label']==label];assert len(values)==4
  cell['labels'][label]={k:{'median':statistics.median(r[k]for r in values),'min':min(r[k]for r in values),'max':max(r[k]for r in values)}for k in ['ns','bytes','allocs']}
 cell['changePercent']={k:100*(cell['labels']['B'][k]['median']/cell['labels']['A'][k]['median']-1)for k in ['ns','bytes','allocs']}
 cell['sameBinaryControlTimePercent']=100*(cell['labels']['D']['ns']['median']/cell['labels']['C']['ns']['median']-1)
 analysis.append(cell)
(e/'comparison-analysis.json').write_text(json.dumps(analysis,indent=2)+'\n')
(e/'validation-receipt.json').write_text(json.dumps({'checks':len(checks),'allPassed':True,'measurements':144,'regressionTopLevelTests':1,'nativeLeafCases':3,'originalExistingPayloadBytes':8388608,'allocationBudgetBytes':12582912,'normalRepeats':1,'raceRepeats':1,'sources':json.loads((e/'environment.json').read_text())['sourceSHA256']},indent=2)+'\n')
for c in analysis:
 a,b=c['labels']['A'],c['labels']['B'];print(c['name'],f"time {a['ns']['median']/1e6:.3f}->{b['ns']['median']/1e6:.3f} ms ({c['changePercent']['ns']:+.2f}%)",f"bytes {a['bytes']['median']}->{b['bytes']['median']}",f"allocs {a['allocs']['median']}->{b['allocs']['median']}",f"AA {c['sameBinaryControlTimePercent']:+.2f}%")
