package fs

import (
 "context"
 "errors"
 "fmt"
 "os"
 "path/filepath"
 "strings"
 "syscall"
 "testing"

 "github.com/chester-hill-solutions/stow-s3/internal/capacity"
 "github.com/chester-hill-solutions/stow-s3/internal/storage"
)

func BenchmarkCapacityFence(b *testing.B) {
 root:=os.Getenv("STOW_BENCH_ROOT")
 if !strings.HasPrefix(root,"/results/") { b.Fatal("Railway volume required") }
 var disk syscall.Statfs_t
 if err:=syscall.Statfs(root,&disk);err!=nil||disk.Type!=61267 { b.Fatalf("ext4 volume required: %v type=%d",err,disk.Type) }
 for _,part:=range []bool{false,true} {
  for _,fault:=range []bool{false,true} {
   b.Run(fmt.Sprintf("part=%t/fault=%t",part,fault),func(b *testing.B){
    body:=strings.Repeat("x",65536)
    b.ReportAllocs();b.ResetTimer()
    for i:=0;i<b.N;i++ {
     b.StopTimer()
     dir,err:=os.MkdirTemp(root,"fence-");if err!=nil { b.Fatal(err) }
     host,err:=capacity.Open(capacity.HostOptions{Dir:filepath.Join(dir,"host"),MaxBytes:64<<20});if err!=nil { b.Fatal(err) }
     ns,err:=host.Bind(capacity.NamespaceOptions{ID:"benchmark",MaxBytes:32<<20,RecoveryReserveBytes:4<<20});if err!=nil { b.Fatal(err) }
     corridor:=filepath.Join(dir,"corridor")
     store,err:=NewFilesystemStoreWithNamespace(filepath.Join(corridor,"store"),ns);if err!=nil { b.Fatal(err) }
     if err:=store.CreateBucket(context.Background(),"bucket");err!=nil { b.Fatal(err) }
     var id string
     if part { upload,err:=store.CreateMultipartUpload(context.Background(),"bucket","key",storage.MultipartOptions{});if err!=nil { b.Fatal(err) };id=upload.UploadID }
     if fault { if err:=os.Chmod(corridor,0300);err!=nil { b.Fatal(err) } }
     b.StartTimer()
     if part { _,err=store.UploadPart(context.Background(),id,1,strings.NewReader(body)) } else { _,err=store.PutObject(context.Background(),"bucket","key",strings.NewReader(body),storage.PutOptions{}) }
     b.StopTimer()
     if fault {
      if !errors.Is(err,os.ErrPermission)||errors.Is(err,storage.ErrMutationCommitted) { b.Fatalf("wrong native failure: %v",err) }
      if (os.Getenv("STOW_BENCH_EXPECT_UNKNOWN")=="1")!=errors.Is(err,capacity.ErrUnknown) { b.Fatalf("wrong reservation outcome: %v",err) }
     } else if err!=nil { b.Fatal(err) }
     if err:=os.Chmod(corridor,0755);err!=nil { b.Fatal(err) }
     snap,err:=ns.Snapshot(context.Background());if err!=nil { b.Fatal(err) }
     pending:=fault&&os.Getenv("STOW_BENCH_EXPECT_UNKNOWN")=="1"
     if pending!=(snap.PendingBytes>0) { b.Fatalf("wrong durable reservation: %+v",snap) }
     if err:=store.Close();err!=nil { b.Fatal(err) }
     if err:=os.RemoveAll(dir);err!=nil { b.Fatal(err) }
     b.StartTimer()
    }
   })
  }
 }
}
