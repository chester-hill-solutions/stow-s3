 "bytes"
 "context"
 "fmt"
 "os"
 "strings"
 "syscall"
 "testing"

 "github.com/chester-hill-solutions/stow-s3/internal/storage"
 "github.com/chester-hill-solutions/stow-s3/internal/capacity"

func BenchmarkCapacityFenceMutation(b *testing.B) {
 root:=os.Getenv("STOW_BENCH_ROOT")
 if !strings.HasPrefix(root,"/results/") { b.Fatal("Railway ext4 fixture required") }
 var disk syscall.Statfs_t
 if err:=syscall.Statfs(root,&disk);err!=nil||disk.Type!=61267 { b.Fatalf("ext4 volume required: %v type=%d",err,disk.Type) }
 for _,objects:=range []int{0,8} {
  for _,op:=range []string{"put","save","delete"} {
   b.Run(fmt.Sprintf("objects=%d/op=%s",objects,op),func(b *testing.B){
    store,ns:=capacityScanFixture(b,root,objects)
    hostdir,_:=ns.Binding();if err:=syncSaveAncestors(hostdir);err!=nil { b.Fatal(err) }
    body:=bytes.Repeat([]byte("y"),64<<10)
    b.ReportAllocs();b.ResetTimer()
    for i:=0;i<b.N;i++ {
     b.StopTimer()
     if op=="delete" { if err:=store.CreateBucket(context.Background(),"empty");err!=nil { b.Fatal(err) } }
     opts:=storage.PutOptions{}
     if op=="save" { opts.RequestKey=fmt.Sprintf("request-%d",i) }
     b.StartTimer()
     var err error
     if op=="delete" { err=store.DeleteBucket(context.Background(),"empty") } else { _,err=store.PutObject(context.Background(),"bucket","target",bytes.NewReader(body),opts) }
     b.StopTimer()
     if err!=nil { b.Fatal(err) }
     b.StartTimer()
    }
    b.StopTimer();assertBenchmarkScanCharge(b,store,ns)
   })
  }
 }
 for _,stage:=range []string{"scan","reconcile"} {
  b.Run("stage="+stage,func(b *testing.B){
   store,ns:=capacityScanFixture(b,root,8)
   hostdir,_:=ns.Binding();if err:=syncSaveAncestors(hostdir);err!=nil { b.Fatal(err) }
   b.ReportAllocs();b.ResetTimer()
   for i:=0;i<b.N;i++ {
    var err error
    if stage=="scan" { _,err=store.scanCapacityLocked() } else { err=store.reconcileCapacityLocked() }
    if err!=nil { b.Fatal(err) }
   }
  })
 }
 b.Run("unbound-create",func(b *testing.B){
  dir,err:=os.MkdirTemp(root,"unbound-");if err!=nil { b.Fatal(err) };b.Cleanup(func(){_ = os.RemoveAll(dir)})
  store,err:=NewFilesystemStore(dir);if err!=nil { b.Fatal(err) };b.Cleanup(func(){_ = store.Close()})
  b.ReportAllocs();b.ResetTimer()
  for i:=0;i<b.N;i++ {
   if err:=store.CreateBucket(context.Background(),"empty");err!=nil { b.Fatal(err) }
   b.StopTimer();if err:=store.DeleteBucket(context.Background(),"empty");err!=nil { b.Fatal(err) };b.StartTimer()
  }
 })
}

func assertBenchmarkScanCharge(b *testing.B,store *FilesystemStore,ns *capacity.Namespace) {
 b.Helper();before,err:=ns.Snapshot(context.Background());if err!=nil { b.Fatal(err) }
 if err:=store.reconcileCapacityLocked();err!=nil { b.Fatal(err) }
 after,err:=ns.Snapshot(context.Background());if err!=nil||before!=after { b.Fatalf("stale charge: before=%+v after=%+v err=%v",before,after,err) }
}
