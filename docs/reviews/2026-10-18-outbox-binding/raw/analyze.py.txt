import json,re,statistics,hashlib,sys

   cell={'depth':depth,'operation':op,'cpu':cpu}
   for variant in ['before','after']:
    items=[x for x in rows if all(x[k]==cell[k]for k in ['depth','operation','cpu'])and x['variant']==variant];assert len(items)==4

   cell['timeChangePercent']=(cell['after']['nsPerOp']['median']/cell['before']['nsPerOp']['median']-1)*100;summary.append(cell)
