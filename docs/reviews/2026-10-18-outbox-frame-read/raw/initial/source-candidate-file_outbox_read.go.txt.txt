package runthrough

import (
	"encoding/binary"
	"fmt"
	"io"
	"math"
	"os"
)

func readOutboxFile(path string) ([]byte, error) {
	file, err := os.Open(path)
	if err != nil {
		return nil, err
	}
	defer file.Close()
	info, err := file.Stat()
	if err != nil {
		return nil, err
	}
	var header [outboxProtoPrefixSize]byte
	n, err := io.ReadFull(file, header[:])
	if err != nil && err != io.EOF && err != io.ErrUnexpectedEOF {
		return nil, err
	}
	if n < len(header) {
		return append([]byte(nil), header[:n]...), nil
	}
	native := string(header[:len(outboxProtoMagic)]) == outboxProtoMagic
	if native && int64(binary.LittleEndian.Uint32(header[len(outboxProtoMagic):]))+outboxProtoPrefixSize != info.Size() {
		return nil, fmt.Errorf("invalid protobuf outbox length")
	}
	return readOutboxRemainder(file, header[:], info.Size(), native)
}

func readOutboxRemainder(file *os.File, prefix []byte, size int64, native bool) ([]byte, error) {
	size = max(size, int64(len(prefix)))
	if size >= math.MaxInt {
		return nil, fmt.Errorf("outbox exceeds platform size limit")
	}
	data := make([]byte, size+1)
	copy(data, prefix)
	n, err := io.ReadFull(file, data[len(prefix):])
	if err == io.EOF || err == io.ErrUnexpectedEOF {
		return data[:len(prefix)+n], nil
	}
	if err != nil {
		return nil, err
	}
	if native {
		return nil, fmt.Errorf("invalid protobuf outbox length")
	}
	tail, err := io.ReadAll(file)
	return append(data, tail...), err
}
