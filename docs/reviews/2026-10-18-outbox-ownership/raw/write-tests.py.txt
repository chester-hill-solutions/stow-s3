var ownershipActions = []string{"renew", "release", "success", "failure", "renew-prepared", "commit-prepared", "discard-prepared"}

type ownershipCredentials struct {
    owner string
    token uint64
}

type ownershipFixture struct {
    path string
    outbox *FileOutbox
}

func newOwnershipFixture(t *testing.T, format, action string, credentials ownershipCredentials) ownershipFixture {
    t.Helper()
    path := filepath.Join(t.TempDir(), "outbox")
    outbox, err := NewFileOutbox(path)
    if err != nil { t.Fatal(err) }
    entry := OutboxEntry{ID: "intent", Operation: OutboxPut, Bucket: "bucket", Key: "key", Version: "original", CreatedAt: time.Unix(100, 0).UTC()}
    if strings.HasSuffix(action, "-prepared") {
        entry.PreparedOwner, entry.PreparedToken = credentials.owner, credentials.token
        _, err = outbox.Prepare(entry)
    } else {
        entry.ClaimOwner, entry.ClaimToken = credentials.owner, credentials.token
        _, err = outbox.Enqueue(entry)
    }
    if err != nil { t.Fatal(err) }
    if format == "json" {
        s := outbox.state
        data, err := json.Marshal(persistedOutbox{Version: outboxLegacyJSONVersion, Entries: s.entries, Prepared: s.prepared, Provenance: s.provenance, Seq: s.seq, NextToken: s.nextToken})
        if err != nil { t.Fatal(err) }
        if err := os.WriteFile(path, data, 0600); err != nil { t.Fatal(err) }
    }
    outbox, err = NewFileOutbox(path)
    if err != nil { t.Fatal(err) }
    return ownershipFixture{path: path, outbox: outbox}
}

func ownershipAction(o *FileOutbox, action, id string, credentials ownershipCredentials) (OutboxEntry, error) {
    switch action {
    case "renew":
        return OutboxEntry{}, o.Renew(id, credentials.owner, credentials.token, time.Hour)
    case "release":
        return OutboxEntry{}, o.Release(id, credentials.owner, credentials.token)
    case "success":
        return OutboxEntry{}, o.MarkClaimedSuccess(id, credentials.owner, credentials.token)
    case "failure":
        return OutboxEntry{}, o.MarkClaimedFailure(id, credentials.owner, credentials.token, errors.New("failed propagation"), time.Unix(4000000000, 0).UTC())
    case "renew-prepared":
        return OutboxEntry{}, o.RenewPrepared(id, credentials.owner, credentials.token, time.Hour)
    case "commit-prepared":
        return o.CommitPrepared(id, credentials.owner, credentials.token, "committed")
    case "discard-prepared":
        return OutboxEntry{}, o.DiscardPreparedOwned(id, credentials.owner, credentials.token)
    default:
        panic("unknown ownership action")
    }
}

func TestOutboxOwnershipRequiresIssuedCredentials(t *testing.T) {
    invalid := []struct { name string; credentials ownershipCredentials }{
        {"unowned", ownershipCredentials{}},
        {"empty-owner", ownershipCredentials{token: 41}},
        {"zero-token", ownershipCredentials{owner: "worker"}},
    }
    for _, format := range []string{"native", "json"} {
        for _, action := range ownershipActions {
            for _, invalid := range invalid {
                t.Run(format+"/"+action+"/"+invalid.name, func(t *testing.T) {
                    f := newOwnershipFixture(t, format, action, invalid.credentials)
                    assertOwnershipRefused(t, f, action, invalid.credentials)
                })
            }
        }
    }
}

func assertOwnershipRefused(t *testing.T, f ownershipFixture, action string, credentials ownershipCredentials) {
    t.Helper()
    before := f.outbox.state.clone()
    data, err := os.ReadFile(f.path)
    if err != nil { t.Fatal(err) }
    got, err := ownershipAction(f.outbox, action, "intent", credentials)
    if !errors.Is(err, ErrOutboxClaimLost) { t.Errorf("non-owner operation accepted: entry=%+v error=%v", got, err) }
    if got != (OutboxEntry{}) { t.Error("refused operation returned a committed intent") }
    actual, err := os.ReadFile(f.path)
    if err != nil { t.Fatal(err) }
    if !bytes.Equal(data, actual) { t.Error("refused operation changed journal bytes") }
    if !reflect.DeepEqual(before, f.outbox.state) { t.Error("refused operation changed in-memory state") }
    peer, err := NewFileOutbox(f.path)
    if err != nil { t.Fatal(err) }
    if !reflect.DeepEqual(before, peer.state) { t.Error("refused operation changed reopened state") }
}

func TestOutboxOwnershipRejectsMismatchedCredentials(t *testing.T) {

                    f := newOwnershipFixture(t, format, action, ownershipCredentials{owner: "worker", token: 41})
                    assertOwnershipRefused(t, f, action, invalid)
                })
            }
        }
    }
}

import (
    "errors"
    "reflect"
    "strings"
    "testing"
    "time"

func TestOutboxOwnershipCurrentOwner(t *testing.T) {
    for _, action := range ownershipActions {
        for _, expiry := range []string{"future", "expired"} {
            t.Run(action+"/"+expiry, func(t *testing.T) { checkCurrentOwnership(t, action, expiry) })
        }
    }
}

func checkCurrentOwnership(t *testing.T, action, expiry string) {
    t.Helper()
    f := newOwnershipFixture(t, "native", action, ownershipCredentials{})
    var entry OutboxEntry
    var err error
    prepared := strings.HasSuffix(action, "-prepared")
    if prepared {
        if err := f.outbox.DiscardPrepared("intent"); err != nil { t.Fatal(err) }
        entry, err = f.outbox.PrepareOwned(OutboxEntry{ID: "intent", Operation: OutboxPut, Bucket: "bucket", Key: "key", Version: "original"}, "worker", time.Hour)
    } else {
        var claimed ClaimedEntry
        var acquired bool
        claimed, acquired, err = f.outbox.Claim("intent", "worker", time.Hour)
        entry = claimed.Entry
        if !acquired { t.Fatal("could not acquire eligible intent") }
    }
    if err != nil { t.Fatal(err) }
    credentials := ownershipCredentials{owner: entry.ClaimOwner, token: entry.ClaimToken}
    if prepared { credentials = ownershipCredentials{owner: entry.PreparedOwner, token: entry.PreparedToken} }
    if credentials.owner == "" || credentials.token == 0 { t.Fatal("allocator returned invalid credentials") }
    if expiry == "expired" {
        err = f.outbox.withState(func(s *outboxState) error {
            if prepared {
                entry.PreparedUntil = time.Unix(100, 0).UTC()
                s.prepared[entry.ID] = entry
            } else {
                entry.ClaimUntil = time.Unix(100, 0).UTC()
                s.entries[entry.ID] = entry
            }
            return nil
        })
        if err != nil { t.Fatal(err) }
    }
    got, err := ownershipAction(f.outbox, action, "intent", credentials)
    if err != nil { t.Fatalf("current owner refused: %v", err) }
    peer, err := NewFileOutbox(f.path)
    if err != nil { t.Fatal(err) }
    verifyOwnershipResult(t, peer.state, action, got)
}

func verifyOwnershipResult(t *testing.T, state outboxState, action string, got OutboxEntry) {
    t.Helper()
    active, activeOK := state.entries["intent"]
    prepared, preparedOK := state.prepared["intent"]
    switch action {
    case "success", "discard-prepared":
        if activeOK || preparedOK { t.Error("completed intent remains queued") }
    case "commit-prepared":
        if !activeOK || preparedOK || active != got || active.Version != "committed" || active.PreparedOwner != "" || active.PreparedToken != 0 { t.Error("owned commit did not publish expected active intent") }
    case "renew":
        if !activeOK || active.ClaimOwner != "worker" || active.ClaimToken == 0 || !active.ClaimUntil.After(time.Now()) { t.Error("claim was not renewed") }
    case "renew-prepared":
        if !preparedOK || prepared.PreparedOwner != "worker" || prepared.PreparedToken == 0 || !prepared.PreparedUntil.After(time.Now()) { t.Error("prepared lease was not renewed") }
    case "release", "failure":
        if !activeOK || active.ClaimOwner != "" || active.ClaimToken != 0 || !active.ClaimUntil.IsZero() { t.Error("claim was not cleared") }
        if action == "failure" && (active.Attempts != 1 || active.LastError != "failed propagation" || active.NextAttempt.IsZero() || active.Terminal) { t.Error("retry fields changed") }
    }
}

func TestOutboxOwnershipMissingAndCommittedRetries(t *testing.T) {
    for _, action := range ownershipActions {
        t.Run(action, func(t *testing.T) {
            f := newOwnershipFixture(t, "native", action, ownershipCredentials{})
            before := f.outbox.state.clone()
            got, err := ownershipAction(f.outbox, action, "missing", ownershipCredentials{})
            noOp := action == "release" || action == "discard-prepared"
            if noOp && err != nil { t.Errorf("missing idempotent operation failed: %v", err) }
            if !noOp && (err == nil || errors.Is(err, ErrOutboxClaimLost)) { t.Errorf("missing intent error changed: %v", err) }
            if got != (OutboxEntry{}) || !reflect.DeepEqual(before, f.outbox.state) { t.Error("missing intent operation changed state") }
        })
    }
    t.Run("already-committed", func(t *testing.T) {
        f := newOwnershipFixture(t, "native", "success", ownershipCredentials{})
        before := f.outbox.state.clone()
        got, err := f.outbox.CommitPrepared("intent", "", 0, "ignored")
        if err != nil || got != before.entries["intent"] || !reflect.DeepEqual(before, f.outbox.state) { t.Fatalf("idempotent committed retry changed: %+v %v", got, err) }
    })
}
