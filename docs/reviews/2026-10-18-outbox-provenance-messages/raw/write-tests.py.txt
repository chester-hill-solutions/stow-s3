import (
    "bytes"

    "time"

func protoBlockFixture(depth int, mixed bool) outboxState {
    s := newOutboxState()
    s.seq, s.nextToken = uint64(depth+1), uint64(depth+1000)
    stamp := time.Date(2026,10,18,1,2,3,123456789,time.UTC)
    for n := 0; n < depth; n++ {
        id := fmt.Sprintf("entry-%06d",n)
        s.provenance[outboxIdentity("bucket",id)] = UpstreamState{ETag:fmt.Sprintf("etag-%d",n),Absent:n%2==0,Unknown:n%3==0}
        if !mixed { continue }
        entry := OutboxEntry{ID:id,Operation:OutboxCopy,Bucket:"bucket",Key:fmt.Sprintf("key-雪-%d",n),SourceBucket:"source",SourceKey:fmt.Sprintf("source-%d",n),Version:fmt.Sprintf("version-%d",n),PreviousVersion:"previous",UpstreamVersion:"upstream",UpstreamAbsent:n%2==0,CreatedAt:stamp,Attempts:n%3,NextAttempt:stamp.Add(time.Hour),LastError:fmt.Sprintf("retry-%d",n),Terminal:n%2==0,ClaimOwner:"claim-owner",ClaimUntil:stamp.Add(time.Minute),ClaimToken:uint64(n+1),Attempted:true,PreparedOwner:"prepared-owner",PreparedUntil:stamp.Add(time.Second),PreparedToken:uint64(n+2)}
        s.entries[id] = entry
        entry.ID = "prepared-"+id
        entry.Prepared = true
        s.prepared[entry.ID] = entry
    }
    return s
}

func TestOutboxProtoBlockRoundTrip(t *testing.T) {
    for _, mixed := range []bool{false,true} {
        for _, depth := range []int{0,1,2,127,128,129,257} {
            t.Run(fmt.Sprintf("mixed=%t/depth=%d",mixed,depth),func(t *testing.T){checkProtoBlockRoundTrip(t,depth,mixed)})
        }
    }
}

func checkProtoBlockRoundTrip(t *testing.T, depth int, mixed bool) {
    t.Helper()
    state := protoBlockFixture(depth,mixed)
    original := state.clone()
    data, err := encodeOutboxProto(state)
    if err != nil { t.Fatal(err) }
    got, err := decodeOutboxProto(data)
    if err != nil || !reflect.DeepEqual(got,state) { t.Fatalf("record values changed: %v",err) }
    again, err := encodeOutboxProto(state)
    if err != nil || !bytes.Equal(data,again) { t.Fatalf("nondeterministic frame: %v",err) }
    if !reflect.DeepEqual(state,original) { t.Fatal("encoding changed caller state") }
    fmt.Printf("STOWPROTOFRAME: {\\"name\\":%q,\\"sha256\\":\\"%x\\",\\"bytes\\":%d}\\n",t.Name(),sha256.Sum256(data),len(data))
}

func TestOutboxProtoBlockConcurrentOwnership(t *testing.T) {
    state := protoBlockFixture(129,true)
    expected, err := encodeOutboxProto(state)
    if err != nil { t.Fatal(err) }
    type result struct { data []byte; err error }
    results := make(chan result,8)
    for n := 0; n < 8; n++ { go func(){data,err:=encodeOutboxProto(state);results<-result{data,err}}() }
    var frames [][]byte
    for n := 0; n < 8; n++ {
        row := <-results
        if row.err != nil || !bytes.Equal(row.data,expected) { t.Fatalf("concurrent frame changed: %v",row.err) }
        frames = append(frames,row.data)
    }
    frames[0][0] ^= 1
    for _, data := range frames[1:] { if !bytes.Equal(data,expected) { t.Fatal("encoded frames share mutable output") } }
}
''')
(r/'internal/runthrough/outbox_proto_blocks_resource_test.go').write_text('''package runthrough

import (
    "fmt"
    "runtime"
    "strings"
    "testing"
    "time"
)

func TestOutboxProtoBlockEarlyErrorAllocation(t *testing.T) {
    state := newOutboxState()
    for n := 0; n < 4096; n++ {
        id:=fmt.Sprintf("entry-%06d",n)
        state.entries[id]=OutboxEntry{ID:id,Operation:OutboxPut,Bucket:"bucket",Key:id}
    }
    invalid:=state.entries["entry-000000"]
    invalid.CreatedAt=time.Date(10000,1,1,0,0,0,0,time.UTC)
    state.entries[invalid.ID]=invalid
    runtime.GC()
    var before, after runtime.MemStats
    runtime.ReadMemStats(&before)
    data, err:=encodeOutboxProto(state)
    runtime.ReadMemStats(&after)
    allocated:=after.TotalAlloc-before.TotalAlloc
    fmt.Printf("STOWPROTOERROR: {\\"allocatedBytes\\":%d,\\"entries\\":4096}\\n",allocated)
    if err==nil || data!=nil {t.Fatal("invalid timestamp produced a frame")}
    if allocated>512<<10 {t.Fatal("early error allocated the whole message collection")}
    runtime.KeepAlive(state)
}

func TestOutboxProtoBlockReleasesInputBuffers(t *testing.T) {
    runtime.GC()
    var before, after runtime.MemStats
    runtime.ReadMemStats(&before)
    frame, err:=encodeProtoBlockStringViews()
    if err!=nil {t.Fatal(err)}
    runtime.GC()
    runtime.ReadMemStats(&after)
    fmt.Printf("STOWPROTOHEAP: {\\"heapBefore\\":%d,\\"heapAfter\\":%d,\\"frameBytes\\":%d}\\n",before.HeapAlloc,after.HeapAlloc,len(frame))
    if after.HeapAlloc>before.HeapAlloc && after.HeapAlloc-before.HeapAlloc>16<<20 {t.Fatal("encoded frame retains caller backing buffers")}
    state, err:=decodeOutboxProto(frame)
    if err!=nil || len(state.entries)!=64 {t.Fatalf("retained frame changed: %v",err)}
    runtime.KeepAlive(frame)
}

func encodeProtoBlockStringViews() ([]byte,error) {
    state:=newOutboxState()
    for n:=0;n<64;n++ {
        value:=fmt.Sprintf("key-%08d",n)+strings.Repeat("x",1<<20)
        id:=fmt.Sprintf("entry-%d",n)
        state.entries[id]=OutboxEntry{ID:id,Operation:OutboxPut,Bucket:"bucket",Key:value[:12],Version:"version"}
    }
    return encodeOutboxProto(state)
}
''')

import (
    "fmt"
    "testing"
)

var protoBlockFrameSink []byte
var protoBlockStateSink outboxState

func BenchmarkOutboxProtoBlocks(b *testing.B) {
    for _,mixed:=range []bool{false,true} {
        for _,depth:=range []int{0,128,4096} {
            for _,operation:=range []string{"encode","decode"} {
                b.Run(fmt.Sprintf("mixed=%t/depth=%d/%s",mixed,depth,operation),func(b *testing.B){benchmarkProtoBlocks(b,depth,mixed,operation)})
            }
        }
    }
}

func benchmarkProtoBlocks(b *testing.B,depth int,mixed bool,operation string) {
    state:=protoBlockFixture(depth,mixed)
    frame,err:=encodeOutboxProto(state)
    if err!=nil {b.Fatal(err)}
    b.ReportAllocs()
    b.ResetTimer()
    for n:=0;n<b.N;n++ {
        if operation=="encode" {protoBlockFrameSink,err=encodeOutboxProto(state)} else {protoBlockStateSink,err=decodeOutboxProto(frame)}
        if err!=nil {b.Fatal(err)}
    }
    b.StopTimer()
    b.ReportMetric(float64(len(frame)),"frame-B")
    if operation=="encode" {frame=protoBlockFrameSink}
    decoded,err:=decodeOutboxProto(frame)
    if err!=nil || len(decoded.provenance)!=depth {b.Fatalf("frame verification: %v",err)}
    if mixed && (len(decoded.entries)!=depth || len(decoded.prepared)!=depth) {b.Fatal("entry count changed")}
    protoBlockFrameSink=nil
    protoBlockStateSink=outboxState{}
}
''')
