 try{const child=spawn('python3',['benchmark/reap_validation.py',binary,...args],{env:{...process.env,TMPDIR:temporaryDirectory},stdio:['ignore',log.fd,log.fd]});result=await new Promise((resolve,reject)=>{child.once('error',reject);child.once('close',(exitCode,signal)=>resolve({exitCode,signal}))})}finally{await log.close()}
 runs.push({cohort,index,variant,binary,binarySHA256:binarySHA256[cohort==='aa'?'after':variant],args,log:name,startedAt,finishedAt:new Date().toISOString(),...result});
 await writeFile(root+'/comparison-runs.json',JSON.stringify({binarySHA256,runs},null,2)+'\n');

}
