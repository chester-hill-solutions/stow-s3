from pathlib import Path
import json,statistics
p=Path('/private/tmp/stow-reconcile-reuse-2026-10-18');j=json.loads((p/'diagnostic-start.json').read_text());e=p/('export-'+j['id']);run=json.loads((e/'run.json').read_text());checks=json.loads((e/'checks.json').read_text());assert run['complete']and run['success']and len(checks)==9 and all(x['passed']for x in checks)
rows=json.loads((e/'measurements.json').read_text());assert len(rows['observations'])==80 and len(rows['commands'])==80 and all(x['iterations']==20 for x in rows['observations'])
import re
for pkg in ['capacity','fs']:assert rows['binaries'][pkg+'-before']==rows['binaries'][pkg+'-after'],'Not the same compiled binary'
trace=(e/'trace.log').read_text();assert '--- FAIL:'not in trace and '--- SKIP:'not in trace
eligibility={}
for op in ['put','save']:
 segment=trace.split('STOW_REUSE_TRACE_BEGIN '+op+'\n')[1].split('STOW_REUSE_TRACE_END '+op+'\n')[0]
 eligibility[op]=re.findall(r'STOW_REUSE_ELIGIBLE (true|false)',segment)

for name in sorted({x['name']for x in rows['observations']}):
 labels={}
 for label in ['A','B','C','D']:
  subset=[x for x in rows['observations']if x['name']==name and x['label']==label];assert len(subset)==4
  labels[label]={k:{'median':statistics.median(x[k]for x in subset),'min':min(x[k]for x in subset),'max':max(x[k]for x in subset)}for k in ['ns','bytes','allocs','maxRSSBytes']}
 c={'name':name,'labels':labels};c['changePercent']={k:100*(labels['B'][k]['median']/labels['A'][k]['median']-1)for k in ['ns','bytes','allocs','maxRSSBytes']};c['sameBinaryControlTimePercent']=100*(labels['D']['ns']['median']/labels['C']['ns']['median']-1);analysis.append(c)
 print(name,{'ns':(labels['A']['ns']['median'],labels['B']['ns']['median']),'bytes':(labels['A']['bytes']['median'],labels['B']['bytes']['median']),'allocs':(labels['A']['allocs']['median'],labels['B']['allocs']['median'])})
(e/'comparison-analysis.json').write_text(json.dumps(analysis,indent=2)+'\n');(e/'validation-receipt.json').write_text(json.dumps({'checks':9,'allPassed':True,'observations':80,'sameBinaryToggle':True,'eligibility':eligibility},indent=2)+'\n')
